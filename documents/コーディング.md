```
const [user, setUser] = useState<User>({
    id: 0,
    number: '',

    email: '',
    img: '',
})

useEffect(() => {
  fetch('/api/users/?number=30305637')
    .then((res) => res.json())

```
- APIのパス
```
app/nextjsapp/src/app/api/users/route.ts
```
- API
```
export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams
  const query = searchParams.get('number')
  let number: string = query ? query : ''

  const user = await prisma.user.findUnique({
    where: {
      number: number,
    },
  })

  return Response.json(user)
}
```

```
export async function GET(request: Request) {  // NextResponse
  const searchParams = request.nextUrl.searchParams // route.js - URL Query Parameters
  const query = searchParams.get('query') // route.js - URL Query Parameters
```
